\section*{Première partie : recherche d'information (8 pts)}

Quelques extraits de recettes de cuisine.

% loup = lait
% sucre = oeuf
% mouton = sucre
%

\begin{tabular}{lp{15cm}}
     $d_1$& Mélanger le lait et trois {\oe}ufs\\
     $d_2$& Assembler deux jaunes d'{\oe}uf et trois blancs d'{\oe}ufs, battre vigoureusement
         et verser l'assemblage des {\oe}ufs dans le moule\\
     $d_3$& Verser la moitié du sucre dans le lait, réserver l'autre moitié dans un bol.\\
     $d_4$&Faire caraméliser le sucre et le recouvrir ensuite avec le lait.\\
     $d_5$&Le pain perdu se fait aisément à base d'{\oe}ufs et de lait.\\ 
\end{tabular}

\smallskip

On va se limiter au vocabulaire \texttt{lait, sucre, {\oe}uf}.

\begin{enumerate}
  \item Donnez la matrice d'incidence \emph{booléenne} (seulement 0 ou 1)
      avec les termes en ligne (NB: on suppose
     une phase préalable de normalisation qui élimine les pluriels, majuscules, etc.)

\begin{solution}
\begin{center}
\begin{tabular}{l|c|c|c|c|c}
           & \textbf{d1} & \textbf{d2} & \textbf{d3} &\textbf{d4}&\textbf{d5} \\ \hline  
   lait    & 1           & 0           & 1           & 1         & 1\\
   sucre  & 0           & 0           & 1           & 1         & 0\\
   {\oe}uf  & 1           & 1           & 0           & 0         & 1\\\hline
   \end{tabular}
\end{center}
\end{solution}
    
  \item Expliquez par quelle technique, avec cette matrice d'incidence, on peut 
     répondre à la requête booléenne "lait et {\oe}uf mais pas sucre".
     Quel est le résultat?
     
\begin{solution}
On fait un \texttt{AND} logique sur les vecteurs "lait" et "{\oe}uf"
et sur la négation du vecteur "sucre".
\end{solution}

   \item Maintenant donnez une matrice d'incidence contenant les tf, 
      et un tableau donnant les idf (sans le $\log$), pour les trois termes précédents.

\begin{solution}
\begin{center}
\begin{tabular}{l|c|c|c|c|c}
            & \textbf{d1} & \textbf{d2} & \textbf{d3} &\textbf{d4}&\textbf{d5} \\ \hline  
   lait  5/4   & 1           & 0           & 1           & 1         & 1\\
   sucre 5/2  & 0           & 0           & 2           & 1         & 0\\
   {\oe}uf 5/3  & 1           & 3           & 0           & 0         & 1\\\hline
   \end{tabular}
\end{center}
\end{solution}

  \item Donner les résultats classés par similarité cosinus basée sur les tf 
  (on ignore l'idf) pour les requêtes suivantes.
  
  \begin{itemize}
    \item {\oe}uf;
    \item lait et sucre;
    \item lait et {\oe}uf.
  \end{itemize}
\begin{solution}
Les normes
\begin{itemize}
  \item $||d_1|| = \sqrt{1+ 1}$; $||d_2|| = \sqrt{3^2}$; $||d_3|| = \sqrt{1+4}$; 
    $||d_4|| = \sqrt{1+1}$, $||d_5|| = \sqrt{1+1}$  
\end{itemize}

Les cosinus (requête non normalisée).

\begin{itemize}
  \item {\oe}uf: d1 : $\frac{1}{\sqrt{2}}$;  
       d2 : $\frac{3}{3}$; 
       d5 : $\frac{1}{\sqrt{2}}$; 
       
       d2 est premier: il ne parle que de {\oe}ufs. d1 et d5
       sont à égalité.
  
  \item lait et sucre: 
  
    d1 : $\frac{1}{\sqrt{2}}$; 
    d3 : $\frac{1+2}{\sqrt{5}}$;
    d4 :  $\frac{1+1}{\sqrt{2}}$  
    d5 : $\frac{1}{\sqrt{2}}$; 
    
    d4 est premier: il a le même équilibre entre les deux termes que la requête.
    d3 est plus orienté sucre que lait, les deux derniers ne parlent
    pas de sucre.
    
  \item lait et {\oe}uf
  
    d1 : $\frac{1+1}{\sqrt{2}}$; 
    d2 : $\frac{3}{3}$;
    d3 :  $\frac{1}{\sqrt{5}}$  
    d4 :  $\frac{1}{\sqrt{2}}$  
    d5 : $\frac{1+1}{\sqrt{2}}$; 
    
  d1 et d5 sont en tête à égalité, devant d2, d4 et d3.
 \end{itemize}

\end{solution}  

   \item On ajoute le document $d_6$: "Une recette sans sucre ni {\oe}uf".
   
   Quel est son score pour la requête "lait et sucre", 
   quel autre document a le même score et qu'est-ce qui change
   si on prend en compte l'idf?
   
\begin{solution}
d6 obtient le même score que d1, mais l'emporte si on considère
l'idf de "sucre" qui est plus élevé  (6/3 en comptant d6) que celui de "lait"
(6/4 en comptant d6).
\end{solution}

   \item Pour la requête "lait et {\oe}uf" et les documents d1 et d5, 
   qu'est-ce qui change si on ne met pas de restriction 
   sur le vocabulaire (tous les mots sont indexés)?
   
\begin{solution}
Dans ce cas le plus petit document (d1) l'emporte car sa norme est
plus petite. Intuititvement: il parle plus \emph{spécifiquement} de
lait et de {\oe}uf que d5.
\end{solution}   
    
  \end{enumerate}
